# Header IP luar: frag HARUS 0, kalau tidak paket masuk ip_defrag dan tidak sampai icmp_rcv.
icmp_ip_header {
	ver_ihl		const[0x45, int8]
	tos		int8
	total_len	len[icmp_ip_packet, int16be]
	id		int16be
	frag		const[0, int16be]
	ttl		int8
	proto		const[1, int8]
	csum		csum[parent, inet, int16be]
	saddr		icmp_src_ip
	daddr		icmp_dst_ip
} [packed]

# Header IP yang dulu dikirim host: src=local, dst=remote
type icmp_inner_ip[PROTO] {
	ver_ihl		const[0x45, int8]
	tos		int8
	total_len	int16be
	id		int16be
	frag		const[0, int16be]
	ttl		int8
	proto		const[PROTO, int8]
	csum		csum[parent, inet, int16be]
	saddr		const[0xac1414aa, int32be]
	daddr		const[0xac1414bb, int32be]
} [packed]

# 8 byte pertama TCP/UDP: sport, dport, lalu seq (TCP) atau len+csum (UDP)
